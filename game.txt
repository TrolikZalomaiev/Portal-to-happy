    <h1>Поймай 20 сердечек ❤️</h1>
    <div class="score-board">Счет: <span id="score">0</span> / 20</div>

   
    <canvas id="gameCanvas" width="350" height="480"></canvas>

   
    <div id="winModal">
        <div class="win-box">
            <h2>Победа! 🎉</h2>
            <p>Ты собрала все 20 сердечек! А теперь с тебя самые теплые и крепкие обнимашки! 🥰</p>
            <button class="btn-restart" onclick="restartGame()">Играть снова</button>
        </div>
    </div>

    <script>
        const canvas = document.getElementById('gameCanvas');
        const ctx = canvas.getContext('2d');
        const scoreEl = document.getElementById('score');
        const winModal = document.getElementById('winModal');

        let score = 0;
        let gameIsOver = false;

        
        let catcher = {
            x: 140,
            y: 410,
            width: 70,
            height: 45
        };

        
        let hearts = [];

        function spawnHeart() {
            if (gameIsOver) return;
            hearts.push({
                x: Math.random() * (canvas.width - 30),
                y: -30,
                size: 28,
                speed: 1.2 + Math.random() * 1.2 
            });
        }

                let heartInterval = setInterval(spawnHeart, 1200);

        
        function moveCatcher(clientX) {
            const rect = canvas.getBoundingClientRect();
            let relativeX = clientX - rect.left - catcher.width / 2;
            if (relativeX < 0) relativeX = 0;
            if (relativeX > canvas.width - catcher.width) relativeX = canvas.width - catcher.width;
            catcher.x = relativeX;
        }

        canvas.addEventListener('mousemove', (e) => moveCatcher(e.clientX));
        canvas.addEventListener('touchmove', (e) => {
            e.preventDefault();
            moveCatcher(e.touches[0].clientX);
        }, { passive: false });

        
        function update() {
            if (gameIsOver) return;

            ctx.clearRect(0, 0, canvas.width, canvas.height);

            // Рисуем корзинку
            ctx.font = '36px Arial';
            ctx.fillText('🧺', catcher.x + 10, catcher.y + 35);

            
            for (let i = hearts.length - 1; i >= 0; i--) {
                let h = hearts[i];
                h.y += h.speed;

                
                ctx.font = `${h.size}px Arial`;
                ctx.fillText('💖', h.x, h.y);

                
                if (
                    h.y + h.size >= catcher.y + 10 &&
                    h.x + h.size/2 >= catcher.x &&
                    h.x + h.size/2 <= catcher.x + catcher.width &&
                    h.y <= catcher.y + catcher.height
                ) {
                    hearts.splice(i, 1);
                    score++;
                    scoreEl.innerText = score;

                    if (score >= 20) {
                        gameIsOver = true;
                        clearInterval(heartInterval);
                        winModal.style.display = 'flex';
                    }
                    continue;
                }

                
                if (h.y > canvas.height) {
                    hearts.splice(i, 1);
                }
            }

            requestAnimationFrame(update);
        }

        function restartGame() {
            score = 0;
            scoreEl.innerText = score;
            hearts = [];
            gameIsOver = false;
            winModal.style.display = 'none';
            clearInterval(heartInterval);
            heartInterval = setInterval(spawnHeart, 1200);
            update();
        }

        
        update();
    </script>
